	cd ~/frappe-bench

	bench --site fjnb enable-scheduler
	# 同步数据库结构
	bench --site fjnb migrate
	# 清理后台缓存
	bench --site fjnb clear-cache
	# 清理网站页面缓存
	bench --site fjnb clear-website-cache

	# 启动后台工人
	bench --site fjnb worker &
	# 启动定时任务调度器
	bench --site fjnb schedule &

	# 查看站点、队列、worker、scheduler 等运行状态
	cd ~/frappe-bench && bench --site fjnb doctor
